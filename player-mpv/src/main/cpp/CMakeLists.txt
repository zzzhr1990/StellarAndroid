cmake_minimum_required(VERSION 3.22)
project(stellar_mpv LANGUAGES CXX)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../../native/dependencies.cmake")
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../../player-source/src/main/cpp" remote-reader)
find_package(player-disc REQUIRED CONFIG)
add_library(stellar_mpv SHARED mpv_jni.cpp remote_stream.cpp http_cache_stream.cpp)
target_compile_features(stellar_mpv PRIVATE cxx_std_17)
target_compile_options(stellar_mpv PRIVATE -Wall -Wextra -Werror)
target_link_libraries(stellar_mpv PRIVATE mpv avformat avcodec avutil stellar_remote_reader smb2 nfs player-disc::stellar_disc android log dl)
target_link_options(stellar_mpv PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
