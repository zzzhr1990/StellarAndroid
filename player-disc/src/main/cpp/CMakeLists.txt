cmake_minimum_required(VERSION 3.22)
project(stellar_disc LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../native/vendor")
add_subdirectory("${VENDOR}/libdvdread" dvdread)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../vendor/libdvdnav" dvdnav)
add_subdirectory("${VENDOR}/libudfread" udfread)
add_subdirectory("${VENDOR}/libbluray" bluray)
add_library(stellar_disc SHARED dvd_reader.cpp bluray_reader.cpp disc_stream.cpp disc_menu.cpp nas_reader.cpp)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../../player-source/src/main/cpp" remote-reader)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../../native/dependencies.cmake")
target_include_directories(stellar_disc PRIVATE include "${VENDOR}/libbluray/src")
target_link_libraries(stellar_disc PRIVATE dvdnav dvdread bluray stellar_remote_reader smb2 nfs)
target_link_options(stellar_disc PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
