cmake_minimum_required(VERSION 3.22)
project(stellar_native LANGUAGES CXX)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../../native/dependencies.cmake")
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../../player-source/src/main/cpp" remote-reader)
add_library(stellar_native SHARED directory_probe.cpp library_io.cpp gif_capture.cpp)
target_compile_features(stellar_native PRIVATE cxx_std_17)
target_compile_options(stellar_native PRIVATE -Wall -Wextra -Werror)
target_link_libraries(stellar_native PRIVATE stellar_remote_reader smb2 nfs avcodec avformat avutil jnigraphics android log)
target_link_options(stellar_native PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
